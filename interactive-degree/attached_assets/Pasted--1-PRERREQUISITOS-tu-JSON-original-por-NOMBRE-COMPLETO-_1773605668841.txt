function splitNameAndCode(full){
  const codeMatch = full.match(/\(([A-Z]{1,3}\d{3,4}(?:\/[A-Z]{1,3}\d{3,4})?)\)\s*$/);
  if (!codeMatch) return { nameOnly: full, code: "" };
  const code = codeMatch[1];
  const nameOnly = full.replace(/\s*\([^)]+\)\s*$/, "");
  return { nameOnly, code };
}

function codeForCourseTitle(title){
  if (CODE_OVERRIDES[title]) return CODE_OVERRIDES[title];
  const { code } = splitNameAndCode(title);
  return code || "";
}

function creditsForCourseTitle(title){
  const code = codeForCourseTitle(title);
  if (!code) return null;
  const cr = CREDITOS_BY_CODE[code];
  return Number.isFinite(cr) ? cr : null;
}

// ---------- helpers ----------
function loadCompleted(){
  try{
    const raw = localStorage.getItem(LS_KEY);
    if(!raw) return new Set();
    const arr = JSON.parse(raw);
    if(!Array.isArray(arr)) return new Set();
    return new Set(arr.filter(x => typeof x === "string"));
  }catch{
    return new Set();
  }
}

function saveCompleted(set){
  localStorage.setItem(LS_KEY, JSON.stringify([...set]));
}

function buildDependents(prereqs){
  const dep = new Map();
  for(const c of Object.keys(prereqs)) dep.set(c, []);
  for(const [course, reqs] of Object.entries(prereqs)){
    for(const r of reqs){
      if(!dep.has(r)) dep.set(r, []);
      dep.get(r).push(course);
    }
  }
  return dep;
}

function isUnlocked(course, completedSet){
  const reqs = PREREQS[course] || [];
  return reqs.every(r => completedSet.has(r));
}

    const x = q.shift();
    const kids = dependents.get(x) || [];
    for(const k of kids){
      if(!seen.has(k)){
        seen.add(k);
        q.push(k);
      }
    }
  }
  seen.delete(course);
  return seen;
}

function immediatePrereqs(course){
  // SOLO los prerequisitos directos
  return new Set(PREREQS[course] || []);
}

function normalizeCompleted(set){
  let changed = true;
  const s = new Set(set);
  while(changed){
    changed = false;
    for(const c of [...s]){
      if(!isUnlocked(c, s) && (PREREQS[c] || []).length > 0){
        s.delete(c);
        changed = true;
      }
    }
  }
  return s;
}

function toast(text){
  elMsg.textContent = text;
  setTimeout(() => {
    if (elMsg.textContent === text) elMsg.textContent = "";
  }, 2200);
}

// ---------- render ----------
function render(justUnlocked){
  courseEls.clear();
  elGrid.innerHTML = "";

  elGrid.classList.add("malla");
  elGrid.style.setProperty("--cols", SEMESTERS.length);

  const total = allCourses.length;
  const done = completed.size;
  elProgress.textContent = `Progreso: ${done}/${total} aprobados`;

  // barras de grupo
  const spans = buildGroupSpans(SEMESTERS);
  for (const s of spans) {
    const bar = document.createElement("div");
    bar.className = "groupBar";
    bar.textContent = s.group;
    bar.style.gridColumn = `${s.start} / span ${s.count}`;
    elGrid.appendChild(bar);
  }

  // columnas
  SEMESTERS.forEach((sem, idx) => {
    const colIndex = idx + 1;

    const semEl = document.createElement("section");
    semEl.className = "semesterCol";
    semEl.style.gridColumn = `${colIndex}`;

    const head = document.createElement("div");
    head.className = "semesterHead";

    const title = document.createElement("p");
    title.className = "title";
    title.textContent = sem.label;

    const semDone = sem.courses.filter(c => completed.has(c)).length;
    const count = document.createElement("div");
    count.className = "count";
    count.textContent = `${semDone}/${sem.courses.length}`;

    head.appendChild(title);
    head.appendChild(count);

    const list = document.createElement("div");
    list.className = "coursesCol";

    for(const course of sem.courses){
      const card = document.createElement("div");
      card.className = "course";

      const done = completed.has(course);
      const unlocked = isUnlocked(course, completed) || done;
      const locked = !unlocked;

      card.classList.toggle("completed", done);
      card.classList.toggle("unlocked", unlocked && !done);
      card.classList.toggle("locked", locked);

      if (justUnlocked.has(course)) card.classList.add("justUnlocked");

      const reqs = PREREQS[course] || [];
      const { nameOnly, code } = splitNameAndCode(course);

      if (reqs.length) card.title = "Requisitos directos:\n- " + reqs.join("\n- ");

      const meta = document.createElement("div");
      meta.className = "courseMeta";

      const codeEl = document.createElement("span");
      codeEl.className = "courseCode";
      codeEl.textContent = code || " ";

      const icon = document.createElement("span");
      icon.className = "courseIcon";
      icon.textContent = done ? "✓" : (locked ? "🔒" : "");

      meta.appendChild(codeEl);
      meta.appendChild(icon);

      const titleEl = document.createElement("p");
      titleEl.className = "courseTitle";
      titleEl.textContent = nameOnly;

      const badges = document.createElement("div");
      badges.className = "badges";

      const b1 = document.createElement("span");
      b1.className = "badge status";
      b1.textContent = done ? "Aprobado" : (locked ? "Bloqueado" : "Disponible");

      const b2 = document.createElement("span");
      b2.className = "badge";
      const cr = creditsForCourseTitle(course);
      b2.textContent = (cr === null) ? "Cr: ?" : `${cr} cr`;

      badges.appendChild(b1);
      badges.appendChild(b2);

      card.appendChild(meta);
      card.appendChild(titleEl);
      card.appendChild(badges);

      card.dataset.course = course;
      courseEls.set(course, card);

      // click toggle
      card.addEventListener("click", () => {
        if (locked && !done) {
          toast("Ese ramo está bloqueado (te faltan prerrequisitos).");
          return;
        }

        const beforeUnlocked = unlockedSet(completed);

        if (done) {
          completed.delete(course);
          const down = downstreamOf(course);
          let removed = 0;
          for (const d of down) if (completed.delete(d)) removed++;
          if (removed > 0) toast(`Se desmarcaron ${removed} ramo(s) dependientes.`);
        } else {
          completed.add(course);
        }

        completed = normalizeCompleted(completed);
        saveCompleted(completed);

        const afterUnlocked = unlockedSet(completed);
        const newly = diff(afterUnlocked, beforeUnlocked);
        render(newly);
      });

      // hover focus (adelante + prereqs directos)
      card.addEventListener("mouseenter", () => focus(course));
      card.addEventListener("mouseleave", clearFocus);

      list.appendChild(card);
    }

    semEl.appendChild(head);
    semEl.appendChild(list);
    elGrid.appendChild(semEl);
  });
}

// spans grupo
function buildGroupSpans(semesters){
  const out = [];
  let start = 1;
  let count = 0;
  let cur = semesters[0]?.group ?? "MALLA";

  semesters.forEach((s, i) => {
    if (s.group !== cur) {
      out.push({ group: cur, start, count });
      cur = s.group;
      start = i + 1;
      count = 1;
    } else {
      count++;
    }
  });

  out.push({ group: cur, start, count });
  return out;
}

function unlockedSet(doneSet){
  const s = new Set();
  for (const c of allCourses) {
    if (isUnlocked(c, doneSet)) s.add(c);
  }
  return s;
}

function diff(a, b){
  const out = new Set();
  for(const x of a) if(!b.has(x)) out.add(x);
  return out;
}

// ---------- hover focus mode ----------
function focus(course){
  const down = downstreamOf(course);
  const prereq = immediatePrereqs(course);

  // texto hover: atrás + adelante
  const nDown = down.size;
  const nPre = prereq.size;

  const preList = nPre ? [...prereq].join(" · ") : "Ninguno";
  const futTxt = nDown === 0 ? "No desbloquea ramos." : `Desbloquea: ${nDown} ramo(s).`;

  elHover.textContent = `Requisitos directos: ${preList} — ${futTxt}`;

  for(const [name, el] of courseEls.entries()){
    el.classList.remove("dim","highlight","focus","prereq");

    if (name === course) el.classList.add("focus");
    else if (prereq.has(name)) el.classList.add("prereq");
    else if (down.has(name)) el.classList.add("highlight");
    else el.classList.add("dim");
  }
}

function clearFocus(){
  elHover.textContent = "";
  for(const el of courseEls.values()){
    el.classList.remove("dim","highlight","focus","prereq");
  }
}
